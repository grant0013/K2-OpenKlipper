cmake_minimum_required(VERSION 3.18)
project(openklipper_touchscreen VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
get_filename_component(SCREEN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include(FetchContent)
FetchContent_Declare(lvgl URL https://codeload.github.com/lvgl/lvgl/tar.gz/refs/tags/v8.4.0 URL_HASH SHA256=3b454d602e1667b9bd2c434f5e8156753d92cf3e71fc40a0cb0d99fca9a605be)
FetchContent_Declare(cjson URL https://codeload.github.com/DaveGamble/cJSON/tar.gz/refs/tags/v1.7.19 URL_HASH SHA256=7fa616e3046edfa7a28a32d5f9eacfd23f92900fe1f8ccd988c1662f30454562)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
FetchContent_GetProperties(lvgl)
if(NOT lvgl_POPULATED)
 FetchContent_Populate(lvgl)
endif()
add_subdirectory(${lvgl_SOURCE_DIR} ${lvgl_BINARY_DIR} EXCLUDE_FROM_ALL)
FetchContent_GetProperties(cjson)
if(NOT cjson_POPULATED)
 FetchContent_Populate(cjson)
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses")
configure_file("${lvgl_SOURCE_DIR}/LICENCE.txt" "${CMAKE_BINARY_DIR}/licenses/LVGL-LICENSE" COPYONLY)
configure_file("${cjson_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/licenses/cJSON-LICENSE" COPYONLY)
file(GLOB UI_FONTS CONFIGURE_DEPENDS "${SCREEN_ROOT}/assets/generated/font_*.c")
if(NOT UI_FONTS)
 message(FATAL_ERROR "Run touchscreen/developer-tools/build_assets.py first (see README)")
endif()
add_executable(openklipper-screen ${SCREEN_ROOT}/src/main.c ${SCREEN_ROOT}/src/platform.c ${SCREEN_ROOT}/src/json.c ${SCREEN_ROOT}/src/view_inputs.c ${SCREEN_ROOT}/src/ui.c ${SCREEN_ROOT}/src/views.c ${SCREEN_ROOT}/src/mesh.c ${SCREEN_ROOT}/src/printer_art.c ${cjson_SOURCE_DIR}/cJSON.c ${UI_FONTS})
target_include_directories(openklipper-screen PRIVATE ${SCREEN_ROOT}/src ${cjson_SOURCE_DIR})
target_link_libraries(openklipper-screen PRIVATE lvgl m pthread)
target_compile_options(openklipper-screen PRIVATE -Wall -Wextra -Wformat=2 -Werror=implicit-function-declaration)
# Keep local checkout, dependency and build paths out of __FILE__ and debug data.
# Apply this to LVGL too: its assertion strings are linked into the runtime.
foreach(target openklipper-screen lvgl)
 target_compile_options(${target} PRIVATE
  "-ffile-prefix-map=${SCREEN_ROOT}=touchscreen"
  "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build"
  "-ffile-prefix-map=${lvgl_SOURCE_DIR}=lvgl"
  "-ffile-prefix-map=${cjson_SOURCE_DIR}=cjson")
endforeach()
option(SCREEN_X11 "Build the interactive X11 simulator" ON)
if(SCREEN_X11 AND NOT CMAKE_CROSSCOMPILING)
 find_package(X11)
 if(X11_FOUND)
  target_compile_definitions(openklipper-screen PRIVATE SCREEN_X11)
  target_link_libraries(openklipper-screen PRIVATE X11::X11)
 endif()
endif()
option(SCREEN_STATIC "Link for the printer without host glibc dependencies" OFF)
if(SCREEN_STATIC)
 target_link_options(openklipper-screen PRIVATE -static)
endif()
install(TARGETS openklipper-screen RUNTIME DESTINATION bin)
install(DIRECTORY ${SCREEN_ROOT}/assets/generated/ DESTINATION share/openklipper-screen/assets PATTERN "*.c" EXCLUDE)
install(DIRECTORY ${SCREEN_ROOT}/backend/ DESTINATION share/openklipper-screen/backend PATTERN "__pycache__" EXCLUDE)
